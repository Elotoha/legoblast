cmake_minimum_required(VERSION 3.24)
project(legoblast_source_port LANGUAGES CXX)

add_library(legoblast_source level.cpp players.cpp sprite.cpp bombs.cpp session.cpp hud.cpp screens.cpp preferences.cpp micro.cpp lia.cpp net_frame.cpp)
target_compile_features(legoblast_source PUBLIC cxx_std_20)
target_include_directories(legoblast_source PUBLIC .)
add_executable(legoblast_levels main.cpp)
target_link_libraries(legoblast_levels PRIVATE legoblast_source)

add_executable(legoblast_lab lab.cpp)
target_link_libraries(legoblast_lab PRIVATE legoblast_source)

option(LEGOBLAST_PLAYER_VIEW "Build the SDL3 player test window" OFF)
if(LEGOBLAST_PLAYER_VIEW)
  set(player_sources player_view.cpp presentation.cpp sound_output.cpp ../src/audio.cpp
      network_panel.cpp netplay.cpp)
  if(ANDROID)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC OFF CACHE BOOL "" FORCE)
    add_subdirectory("${LEGOBLAST_ANDROID_DEPS}/SDL3-3.2.30" "${CMAKE_CURRENT_BINARY_DIR}/sdl")
    set_target_properties(legoblast_source PROPERTIES POSITION_INDEPENDENT_CODE ON)
    add_library(legoblast_players SHARED ${player_sources} ../android/audio_io.cpp ../android/net_socket.cpp)
    target_include_directories(legoblast_players PRIVATE "${LEGOBLAST_ANDROID_DEPS}")
    target_compile_options(legoblast_source PRIVATE -O2)
    target_compile_options(legoblast_players PRIVATE -O2)
  else()
    find_package(SDL3 3.2 CONFIG REQUIRED)
    find_package(SndFile CONFIG QUIET)
    find_package(CURL 8.0 REQUIRED)
    add_executable(legoblast_players ${player_sources} ../src/audio_io.cpp net_socket_curl.cpp)
    target_link_libraries(legoblast_players PRIVATE CURL::libcurl)
    if(TARGET SndFile::sndfile)
      target_link_libraries(legoblast_players PRIVATE SndFile::sndfile)
    else()
      find_path(SNDFILE_INCLUDE_DIR sndfile.h REQUIRED)
      find_library(SNDFILE_LIBRARY NAMES sndfile libsndfile.so.1 REQUIRED)
      target_include_directories(legoblast_players PRIVATE "${SNDFILE_INCLUDE_DIR}")
      target_link_libraries(legoblast_players PRIVATE "${SNDFILE_LIBRARY}")
    endif()
  endif()
  target_include_directories(legoblast_players PRIVATE ../src)
  target_compile_definitions(legoblast_players PRIVATE LEGOBLAST_AUDIO_PATH="${CMAKE_CURRENT_SOURCE_DIR}/audio")
  target_link_libraries(legoblast_players PRIVATE legoblast_source SDL3::SDL3)
endif()
