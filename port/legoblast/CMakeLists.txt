cmake_minimum_required(VERSION 3.24)
project(legoblast_source_port LANGUAGES CXX)

add_library(legoblast_source level.cpp players.cpp sprite.cpp)
target_compile_features(legoblast_source PUBLIC cxx_std_20)
target_include_directories(legoblast_source PUBLIC .)
add_executable(legoblast_levels main.cpp)
target_link_libraries(legoblast_levels PRIVATE legoblast_source)

option(LEGOBLAST_PLAYER_VIEW "Build the SDL3 player test window" OFF)
if(LEGOBLAST_PLAYER_VIEW)
  find_package(SDL3 3.2 CONFIG REQUIRED)
  add_executable(legoblast_players player_view.cpp)
  target_link_libraries(legoblast_players PRIVATE legoblast_source SDL3::SDL3)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(legoblast_player_checks player_checks.cpp)
  target_link_libraries(legoblast_player_checks PRIVATE legoblast_source)
  add_test(NAME source_players COMMAND legoblast_player_checks)
  add_executable(legoblast_source_checks checks.cpp)
  target_link_libraries(legoblast_source_checks PRIVATE legoblast_source)
  add_test(NAME source_port COMMAND legoblast_source_checks "${CMAKE_CURRENT_SOURCE_DIR}/../../amiga/src")
endif()
