cmake_minimum_required(VERSION 3.24)
project(legoblast_source_port LANGUAGES CXX)

add_library(legoblast_source level.cpp players.cpp sprite.cpp bombs.cpp session.cpp hud.cpp screens.cpp preferences.cpp micro.cpp lia.cpp)
target_compile_features(legoblast_source PUBLIC cxx_std_20)
target_include_directories(legoblast_source PUBLIC .)
add_executable(legoblast_levels main.cpp)
target_link_libraries(legoblast_levels PRIVATE legoblast_source)

add_executable(legoblast_lab lab.cpp)
target_link_libraries(legoblast_lab PRIVATE legoblast_source)

option(LEGOBLAST_PLAYER_VIEW "Build the SDL3 player test window" OFF)
if(LEGOBLAST_PLAYER_VIEW)
  find_package(SDL3 3.2 CONFIG REQUIRED)
  find_package(SndFile CONFIG QUIET)
  add_executable(legoblast_players player_view.cpp sound_output.cpp ../src/audio.cpp ../src/audio_io.cpp)
  target_include_directories(legoblast_players PRIVATE ../src)
  target_compile_definitions(legoblast_players PRIVATE LEGOBLAST_AUDIO_PATH="${CMAKE_CURRENT_SOURCE_DIR}/audio")
  target_link_libraries(legoblast_players PRIVATE legoblast_source SDL3::SDL3)
  if(TARGET SndFile::sndfile)
    target_link_libraries(legoblast_players PRIVATE SndFile::sndfile)
  else()
    find_path(SNDFILE_INCLUDE_DIR sndfile.h REQUIRED)
    find_library(SNDFILE_LIBRARY NAMES sndfile libsndfile.so.1 REQUIRED)
    target_include_directories(legoblast_players PRIVATE "${SNDFILE_INCLUDE_DIR}")
    target_link_libraries(legoblast_players PRIVATE "${SNDFILE_LIBRARY}")
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(legoblast_lia_checks lia_checks.cpp)
  target_link_libraries(legoblast_lia_checks PRIVATE legoblast_source)
  add_test(NAME lia_short_fuse COMMAND legoblast_lia_checks)
  add_executable(legoblast_session_checks session_checks.cpp)
  target_link_libraries(legoblast_session_checks PRIVATE legoblast_source)
  add_test(NAME source_session COMMAND legoblast_session_checks)
  add_executable(legoblast_bomb_checks bomb_checks.cpp)
  target_link_libraries(legoblast_bomb_checks PRIVATE legoblast_source)
  add_test(NAME source_bombs COMMAND legoblast_bomb_checks)
  add_executable(legoblast_player_checks player_checks.cpp)
  target_link_libraries(legoblast_player_checks PRIVATE legoblast_source)
  add_test(NAME source_players COMMAND legoblast_player_checks)
  add_executable(legoblast_source_checks checks.cpp)
  target_link_libraries(legoblast_source_checks PRIVATE legoblast_source)
  add_test(NAME source_port COMMAND legoblast_source_checks "${CMAKE_CURRENT_SOURCE_DIR}/../../amiga/src")
endif()
